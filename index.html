function doGet(e) { var output = ContentService.createTextOutput(); output.setMimeType(ContentService.MimeType.JSON); try { var ss = SpreadsheetApp.openByUrl("https://docs.google.com/spreadsheets/d/1VcZ1G1r6chHgn9hsX0OXZNGWUw4rfuuKdOXNd98SPZk/edit"); // ========================================== // 功能一：處理「打卡修正申請」 (保持不變) // ========================================== if (e.parameter.type === "correction") { var correctionSheet = ss.getSheetByName("打卡修正申請"); if (!correctionSheet) { return output.setContent(JSON.stringify({ "status": "error", "message": "找不到「打卡修正申請」分頁，請聯絡管理員建立。" })); } var now = new Date(); var lineName = e.parameter.name || "未知員工"; var actionType = e.parameter.action; var reqDate = e.parameter.reqDate; var reqTime = e.parameter.reqTime; var reason = e.parameter.reason; correctionSheet.appendRow([now, lineName, actionType, reqDate, reqTime, reason, "待審核"]); return output.setContent(JSON.stringify({ "status": "success", "message": "修正申請已送出，請等候主管審核！" })); } // ========================================== // 功能二：處理「特休申請」 (5欄位，無事由) (保持不變) // ========================================== if (e.parameter.type === "leave") { var leaveSheet = ss.getSheetByName("特休申請"); if (!leaveSheet) { return output.setContent(JSON.stringify({ "status": "error", "message": "找不到「特休申請」分頁，請聯絡管理員建立。" })); } var now = new Date(); var lineName = e.parameter.name || "未知員工"; // 依序寫入 A 到 E 欄 leaveSheet.appendRow([ now, // A欄：申請時間 lineName, // B欄：員工姓名 e.parameter.leaveDate, // C欄：特休日期 e.parameter.leaveHours, // D欄：申請時數 "待審核" // E欄：審核狀態 ]); return output.setContent(JSON.stringify({ "status": "success", "message": "特休申請已送出，請等候主管審核！" })); } // ========================================== // 功能三：處理「正常打卡」 (包含【以天計算】防呆機制與特殊事項備註) // ========================================== if (e.parameter.action && !e.parameter.type) { var sheet = ss.getSheets()[0]; // 寫入第一個分頁 (主打卡紀錄) var lineName = e.parameter.name || "未知員工"; var actionType = e.parameter.action; // "上班" 或 "下班" var lat = e.parameter.lat; var lng = e.parameter.lng; var remark = e.parameter.remark || ""; // 抓取特殊事項備註 var now = new Date(); var todayStr = now.toDateString(); // 取得今天的日期字串 (用來判斷是否跨日) var data = sheet.getDataRange().getValues(); var lastActionToday = ""; // 只記錄「今天」的最後一次動作 var lastRecordTime = null; // 記錄最後一次打卡的時間 (不限日期，用來算3分鐘冷卻) // 1. 找出該員工的「上一筆」打卡紀錄，並判斷是不是今天的紀錄 for (var i = data.length - 1; i >= 1; i--) { if (data[i][1] === lineName) { lastRecordTime = new Date(data[i][0]); // 如果這筆紀錄是「今天」的，才把動作記錄下來 if (lastRecordTime.toDateString() === todayStr) { lastActionToday = data[i][2]; } break; // 找到最近一筆就停止迴圈 } } // 2. 防呆機制：檢查「今天」是否連續打相同的動作 if (lastActionToday === actionType) { var errorMsg = (actionType === "上班") ? "您今日已經打過「上班」了，請先進行「下班」打卡！" : "您今日已經打過「下班」了，請勿重複打卡！"; return output.setContent(JSON.stringify({ "status": "error", "message": errorMsg })); } // 3. 防呆機制：如果是今天第一次打卡，卻直接按「下班」 if (lastActionToday === "" && actionType === "下班") { return output.setContent(JSON.stringify({ "status": "error", "message": "查無您【今日】的上班紀錄，無法直接進行下班打卡。" })); } // 4. 防呆機制：3 分鐘內無法重複打卡 (維持原本邏輯) if (lastRecordTime) { var diffInSeconds = (now - lastRecordTime) / 1000; if (diffInSeconds < 180) { return output.setContent(JSON.stringify({ "status": "error", "message": lineName + "已打卡，三分鐘內無法重複打卡" })); } } // 5. 通過所有檢查，寫入正常打卡資料 (第六欄放入備註) sheet.appendRow([now, lineName, actionType, lat, lng, remark]); return output.setContent(JSON.stringify({ "status": "success", "message": "【" + actionType + "】打卡成功！" })); } return output.setContent(JSON.stringify({ "status": "ready", "message": "打卡系統後端正常運作中" })); } catch (error) { return output.setContent(JSON.stringify({ "status": "error", "message": error.message })); } }
